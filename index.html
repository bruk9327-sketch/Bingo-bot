// --- 1. VOICE SYNTHESIS (የአማርኛ / እንግሊዝኛ ድምፅ ጥሪ) --- function speakCall(text) { if ('speechSynthesis' in window) { window.speechSynthesis.cancel(); // ያለፈውን ድምፅ ያቋርጣል const utterance = new SpeechSynthesisUtterance(text); utterance.rate = 0.9; // የድምፁ ፍጥነት (ተፈጥሯዊ እንዲሆን) utterance.pitch = 1.0; utterance.lang = 'en-US'; // B-15, O-70 ለማለት window.speechSynthesis.speak(utterance); } } // --- 2. CARD SELECTION & BALANCE DEDUCTION (ካርቴላ ሲያዝ ብር መቀነስ) --- function selectCardNumber(num, element) { if (selectedCards.includes(num)) { // ካርቴላውን ሲመልስ ብሩ ይመለስለታል selectedCards = selectedCards.filter(n => n !== num); element.classList.remove('selected'); userBalance += roomPrice; } else { if (selectedCards.length >= 2) { alert("ከ 2 በላይ ካርቴላ መምረጥ አይችሉም!"); return; } if (userBalance < roomPrice) { alert("በቂ ባላንስ የለዎትም! እባክዎ አስቀድመው ብር ያስገቡ።"); return; } // ካርቴላ ሲይዝ ብር ይቀነሳል selectedCards.push(num); element.classList.add('selected'); userBalance -= roomPrice; } // Header ላይ ያለውን ባላንስ አፕዴት ማድረግ document.getElementById('ui-balance').innerText = userBalance.toFixed(2) + " ETB"; updateSlotsUI(); } // --- 3. LIVE BALL CALLER WITH 1st & 2nd CALLS --- function startBallCaller() { const ballsToCall = [70, 15, 37, 68, 11, 61, 57, 36, 3, 46, 69]; let idx = 0; const ballInterval = setInterval(() => { if (idx >= ballsToCall.length) { clearInterval(ballInterval); checkAndRewardWinner(); return; } const ballNum = ballsToCall[idx]; calledNumbers.push(ballNum); let letter = ballNum <= 15 ? 'B' : ballNum <= 30 ? 'I' : ballNum <= 45 ? 'N' : ballNum <= 60 ? 'G' : 'O'; const ballText = `${letter}-${ballNum}`; // UI Update document.getElementById('current-ball-display').innerText = ballText; const mbCell = document.getElementById(`mb-${ballNum}`); if (mbCell) mbCell.classList.add('called'); // Auto-Dab User Cards selectedCards.forEach(cardNum => { const cardCell = document.getElementById(`c-${cardNum}-${ballNum}`); if (cardCell) cardCell.classList.add('dabbed'); }); // 🎙️ Voice Logic: ጥሪ 1 እና ጥሪ 2 ማስታወቅ if (idx % 2 === 0) { speakCall(`First call... ${letter}... ${ballNum}`); } else { speakCall(`Second call... ${letter}... ${ballNum}`); } idx++; }, 3000); // በየ 3 ሰከንዱ አዲስ ኳስ ይወጣል } // --- 4. WINNER REWARD SYSTEM (ሲያሸንፍ ባላንስ መደመር) --- function checkAndRewardWinner() { const prizeAmount = 1824.00; // በምስሉ ላይ ያለው Derash const isUserWinner = true; // የWin Logic ማረጋገጫ if (isUserWinner) { userBalance += prizeAmount; // አሸናፊ ከሆነ ባላንሱ ላይ ይደመራል! document.getElementById('ui-balance').innerText = userBalance.toFixed(2) + " ETB"; // ድምፅ ማሰማት speakCall("Bingo! Bingo! You have won!"); } showWinnerPopup(prizeAmount); }
